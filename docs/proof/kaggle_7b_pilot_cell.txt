# ── 7B PILOT ─────────────────────────────────────────────────────────────
# Paste as a NEW cell at the end of the Kaggle notebook and run it.
# Self-contained: finds (or unpacks) the project itself, so it does not
# matter which earlier cells ran in this session.
# Cost: ~1 hour on a T4. Writes only artifacts/results/phase3_pilot.json.

import glob, json, os, shutil, subprocess, sys, zipfile
from pathlib import Path

WORK, INPUT = Path('/kaggle/working'), Path('/kaggle/input')
PROJECT = WORK / 'project'

# 1. locate the project root - the directory holding configs/default.yaml.
#    A Kaggle dataset is versioned and a notebook stays pinned to the version it was
#    attached at, so an old copy can easily still be sitting in /kaggle/input after an
#    upload. Prefer a tree that actually contains the fix, and say so if none does.
def _has_fix(root):
    cfg = Path(root) / "src" / "adaptive_reasoning" / "config.py"
    return cfg.exists() and "device_map" in cfg.read_text(encoding="utf-8")


candidates = []
for marker in glob.glob(str(WORK / '**' / 'configs' / 'default.yaml'), recursive=True):
    candidates.append(Path(marker).parent.parent)

if not candidates:
    zips = glob.glob(str(INPUT / '**' / 'adaptive-reasoning.zip'), recursive=True)
    marks = glob.glob(str(INPUT / '**' / 'configs' / 'default.yaml'), recursive=True)
    if zips:
        print(f'unpacking {zips[0]}')
        with zipfile.ZipFile(zips[0]) as zf:
            zf.extractall(WORK)
        candidates = [Path(m).parent.parent for m in
                      glob.glob(str(WORK / '**' / 'configs' / 'default.yaml'), recursive=True)]
    elif marks:
        sources = [Path(m).parent.parent for m in marks]
        pick = next((s for s in sources if _has_fix(s)), sources[0])
        print(f'copying project from {pick}')
        if PROJECT.exists():
            shutil.rmtree(PROJECT)
        shutil.copytree(pick, PROJECT)
        candidates = [PROJECT]
    else:
        raise SystemExit('No project found. Attach the adaptive-reasoning-fas dataset '
                         'in the sidebar under Input, then re-run this cell.')

root = next((c for c in candidates if _has_fix(c)), None)
if root is None:
    print('Found a project at:', candidates[0])
    raise SystemExit(
        """
        This is an OLD version of the dataset - it has no llm.device_map, so a 7B
        model cannot be loaded (it would be built in CPU RAM and stall at ~42%).

        Kaggle pins a notebook to the dataset version it was attached at, so a new
        upload does not move it. To fix:

          1. sidebar -> Input -> remove adaptive-reasoning-fas
          2. Add Input -> your Datasets -> adaptive-reasoning-fas (latest version)
          3. re-run this cell
        """
    )
print('using project:', root)

os.chdir(root)
print('working directory:', os.getcwd())
assert Path('scripts/run_pilot.py').exists(), 'scripts/run_pilot.py missing under ' + str(root)

# 2. dependencies - only install if this is a fresh session
try:
    import transformers, torch          # noqa: F401
    print('dependencies already present')
except ImportError:
    print('installing dependencies (a few minutes) ...')
    subprocess.check_call([sys.executable, '-m', 'pip', 'install', '-q',
                           '-r', 'requirements.txt'])
    subprocess.check_call([sys.executable, '-m', 'pip', 'install', '-q', '-e', '.'])

# 3. the 7B config.
#    device_map=auto is the critical line. Without it transformers builds the whole
#    model in CPU RAM (~14 GB for 7B in float16) before moving it to the GPU, and a
#    Kaggle session has about 13 GB - the load stalls part-way through "Materializing
#    param" and never recovers. With it, shards stream straight to the two T4s.
Path('configs/experiment').mkdir(parents=True, exist_ok=True)
Path('configs/experiment/pilot7b.yaml').write_text("""llm:
  model_id: deepseek-ai/DeepSeek-R1-Distill-Qwen-7B
  device_map: auto
  batch_size: 4
  max_new_tokens: 1536

traces:
  step_tokens: 48
  max_steps: 32
  probe_max_tokens: 16
  pilot:
    n_questions: 50
""")

rc = subprocess.call([sys.executable, 'scripts/run_pilot.py', '--experiment', 'pilot7b'])
print('\nexit code:', rc)

# 4. the numbers that decide the project
out = Path('artifacts/results/phase3_pilot.json')
if rc != 0 or not out.exists():
    raise SystemExit(f'Pilot did not finish (exit {rc}). Read the error above - the '
                     'usual causes are out-of-memory (lower batch_size to 2) or '
                     'Internet being off in the sidebar.')

p = json.loads(out.read_text())
print('\n' + '=' * 58)
print('  7B PILOT vs 1.5B BASELINE')
print('=' * 58)
print(f"  accuracy            {p['accuracy']*100:5.1f}%      (1.5B was 52.0%)")
print(f"  median tokens       {p['median_reasoning_tokens']:5.0f}       (1.5B was 524)")
print(f"  truncation rate     {p['truncation_rate']*100:5.1f}%      (1.5B was 12.0%)")
print(f"  tokens/second       {p['tokens_per_second']:5.1f}       (1.5B was 91.5)")
print(f"  projected full run  {p['projected_phase3_hours']:5.1f} hours")
print(f"  gate passed         {p['passed']}")
print('=' * 58)
print('\nDECISION RULE')
print('  accuracy >= 65%  -> run the full 7B trace generation, 60-70% is reachable')
print('  accuracy 55-65%  -> borderline; worth one full run if GPU quota allows')
print('  accuracy <  55%  -> stay on 1.5B and spend the time on phases 7-8')
