# ── 7B PILOT ─────────────────────────────────────────────────────────────
# Paste as a NEW cell at the end of the Kaggle notebook and run it.
#
# Data comes from the attached dataset; CODE is pulled fresh from GitHub each
# run. That split matters: a Kaggle notebook stays pinned to the dataset
# version it was attached at, so re-uploading code as a dataset does not reach
# the notebook, while a git clone always gets the current commit.
#
# Cost: ~1 hour on a T4. Writes only artifacts/results/phase3_pilot.json.

import glob, json, os, shutil, subprocess, sys
from pathlib import Path

REPO = 'https://github.com/KJSK-Koushik/Adaptive-reasoning-for-Financial-Advisory-Systems.git'
WORK, INPUT = Path('/kaggle/working'), Path('/kaggle/input')
PROJECT, FRESH = WORK / 'project', WORK / '_repo'

# ── 1. data, from the attached dataset ───────────────────────────────────
found = glob.glob(str(INPUT / '**' / 'data' / 'processed' / 'unified.parquet'),
                  recursive=True)
if not found:
    raise SystemExit(
        'unified.parquet not found. Sidebar -> Add Input -> your Datasets -> '
        'adaptive-reasoning-fas, then re-run this cell.'
    )
data_root = Path(found[0]).parent.parent.parent
print('data from:', data_root)

if PROJECT.exists():
    shutil.rmtree(PROJECT)
PROJECT.mkdir(parents=True)
shutil.copytree(data_root / 'data', PROJECT / 'data')

# ── 2. code, from GitHub ─────────────────────────────────────────────────
if FRESH.exists():
    shutil.rmtree(FRESH)
print('cloning the current code ...')
subprocess.check_call(['git', 'clone', '--depth', '1', '--quiet', REPO, str(FRESH)])

for part in ('src', 'scripts', 'configs'):
    shutil.copytree(FRESH / part, PROJECT / part)
for f in ('requirements.txt', 'requirements-gpu.txt', 'pyproject.toml'):
    if (FRESH / f).exists():
        shutil.copy2(FRESH / f, PROJECT / f)

commit = subprocess.run(['git', '-C', str(FRESH), 'log', '-1', '--format=%h %s'],
                        capture_output=True, text=True).stdout.strip()
print('code at commit:', commit)

os.chdir(PROJECT)
print('working directory:', os.getcwd())

# The fix this pilot depends on. Without device_map, transformers builds the whole
# model in CPU RAM (~14 GB for 7B in float16) before moving it to the GPU, and a
# Kaggle session has about 13 GB - the load stalls part-way through "Materializing
# param" and never recovers.
cfg_src = Path('src/adaptive_reasoning/config.py').read_text(encoding='utf-8')
assert 'device_map' in cfg_src, 'cloned code has no device_map - check the repo'
print('device_map supported: True')

# ── 3. dependencies ──────────────────────────────────────────────────────
try:
    import torch, transformers  # noqa: F401
    print('dependencies already present')
except ImportError:
    print('installing dependencies (a few minutes) ...')
    subprocess.check_call([sys.executable, '-m', 'pip', 'install', '-q',
                           '-r', 'requirements.txt'])

# ── 4. the 7B config ─────────────────────────────────────────────────────
Path('configs/experiment').mkdir(parents=True, exist_ok=True)
Path('configs/experiment/pilot7b.yaml').write_text("""llm:
  model_id: deepseek-ai/DeepSeek-R1-Distill-Qwen-7B
  device_map: auto
  batch_size: 4
  max_new_tokens: 1536

traces:
  step_tokens: 48
  max_steps: 32
  probe_max_tokens: 16
  pilot:
    n_questions: 50
""")

rc = subprocess.call([sys.executable, 'scripts/run_pilot.py', '--experiment', 'pilot7b'])
print('\nexit code:', rc)

# ── 5. the numbers that decide the project ───────────────────────────────
out = Path('artifacts/results/phase3_pilot.json')
if rc != 0 or not out.exists():
    raise SystemExit(
        f'Pilot did not finish (exit {rc}). Read the error above - the usual causes '
        'are out-of-memory (lower batch_size to 2) or Internet being off in the '
        'sidebar.'
    )

p = json.loads(out.read_text())
print('\n' + '=' * 58)
print('  7B PILOT vs 1.5B BASELINE')
print('=' * 58)
print(f"  accuracy            {p['accuracy']*100:5.1f}%      (1.5B was 52.0%)")
print(f"  median tokens       {p['median_reasoning_tokens']:5.0f}       (1.5B was 524)")
print(f"  truncation rate     {p['truncation_rate']*100:5.1f}%      (1.5B was 12.0%)")
print(f"  tokens/second       {p['tokens_per_second']:5.1f}       (1.5B was 91.5)")
print(f"  projected full run  {p['projected_phase3_hours']:5.1f} hours")
print(f"  gate passed         {p['passed']}")
print('=' * 58)
print('\nDECISION RULE')
print('  accuracy >= 65%  -> run the full 7B trace generation, 60-70% is reachable')
print('  accuracy 55-65%  -> borderline; worth one full run if GPU quota allows')
print('  accuracy <  55%  -> stay on 1.5B and spend the time on phases 7-8')
