# ── 7B PILOT ─────────────────────────────────────────────────────────────
# Paste as a NEW cell at the end of the Kaggle notebook and run it.
# Self-contained: finds the project itself, and refuses to run against an old
# copy of the dataset rather than failing obscurely ten seconds later.
# Cost: ~1 hour on a T4. Writes only artifacts/results/phase3_pilot.json.

import glob, json, os, shutil, subprocess, sys, zipfile
from pathlib import Path

WORK, INPUT = Path('/kaggle/working'), Path('/kaggle/input')
PROJECT = WORK / 'project'


def has_fix(root):
    """Does this project tree support llm.device_map?

    Without it, transformers builds the whole model in CPU RAM (~14 GB for 7B in
    float16) before moving it to the GPU. A Kaggle session has about 13 GB, so the
    load stalls part-way through "Materializing param" and never recovers.
    """
    cfg = Path(root) / 'src' / 'adaptive_reasoning' / 'config.py'
    return cfg.exists() and 'device_map' in cfg.read_text(encoding='utf-8')


def roots_under(base):
    """Every project tree beneath `base`, newest-looking first."""
    found = glob.glob(str(base / '**' / 'configs' / 'default.yaml'), recursive=True)
    return [Path(m).parent.parent for m in found]


# 1. find a project tree that has the fix, looking in the working copy first and
#    then in the attached dataset. A stale /kaggle/working/project left by an
#    earlier run must not shadow a freshly attached dataset version.
candidates = roots_under(WORK) + roots_under(INPUT)

if not candidates:
    zips = glob.glob(str(INPUT / '**' / 'adaptive-reasoning.zip'), recursive=True)
    if not zips:
        raise SystemExit(
            'No project found. Sidebar -> Add Input -> your Datasets -> '
            'adaptive-reasoning-fas, then re-run this cell.'
        )
    print('unpacking', zips[0])
    with zipfile.ZipFile(zips[0]) as zf:
        zf.extractall(WORK)
    candidates = roots_under(WORK)

fixed = [c for c in candidates if has_fix(c)]
if not fixed:
    print('project trees found:')
    for c in candidates:
        print('   ', c)
    raise SystemExit(
        """
        All of these are an OLD version of the code - none supports llm.device_map,
        so the 7B model cannot be loaded.

        Kaggle pins a notebook to the dataset version it was attached at, so
        uploading a new version does not move it. To fix:

          1. sidebar -> Input -> remove adaptive-reasoning-fas
          2. Add Input -> your Datasets -> adaptive-reasoning-fas (latest version)
          3. Run -> Factory reset  (clears the stale /kaggle/working copy)
          4. re-run this cell
        """
    )

source = fixed[0]

# 2. /kaggle/input is read-only, so work from a copy under /kaggle/working.
if str(source).startswith(str(INPUT)):
    if PROJECT.exists():
        print('replacing the stale copy at', PROJECT)
        shutil.rmtree(PROJECT)
    print('copying project from', source)
    shutil.copytree(source, PROJECT)
    root = PROJECT
else:
    root = source

os.chdir(root)
print('working directory:', os.getcwd())
assert Path('scripts/run_pilot.py').exists(), f'run_pilot.py missing under {root}'
print('device_map supported: True')

# 3. dependencies - only install if this is a fresh session
try:
    import torch, transformers  # noqa: F401
    print('dependencies already present')
except ImportError:
    print('installing dependencies (a few minutes) ...')
    subprocess.check_call([sys.executable, '-m', 'pip', 'install', '-q',
                           '-r', 'requirements.txt'])
    subprocess.check_call([sys.executable, '-m', 'pip', 'install', '-q', '-e', '.'])

# 4. the 7B config. device_map=auto streams shards straight to the two T4s.
Path('configs/experiment').mkdir(parents=True, exist_ok=True)
Path('configs/experiment/pilot7b.yaml').write_text("""llm:
  model_id: deepseek-ai/DeepSeek-R1-Distill-Qwen-7B
  device_map: auto
  batch_size: 4
  max_new_tokens: 1536

traces:
  step_tokens: 48
  max_steps: 32
  probe_max_tokens: 16
  pilot:
    n_questions: 50
""")

rc = subprocess.call([sys.executable, 'scripts/run_pilot.py', '--experiment', 'pilot7b'])
print('\nexit code:', rc)

# 5. the numbers that decide the project
out = Path('artifacts/results/phase3_pilot.json')
if rc != 0 or not out.exists():
    raise SystemExit(
        f'Pilot did not finish (exit {rc}). Read the error above - the usual causes '
        'are out-of-memory (lower batch_size to 2) or Internet being off in the '
        'sidebar.'
    )

p = json.loads(out.read_text())
print('\n' + '=' * 58)
print('  7B PILOT vs 1.5B BASELINE')
print('=' * 58)
print(f"  accuracy            {p['accuracy']*100:5.1f}%      (1.5B was 52.0%)")
print(f"  median tokens       {p['median_reasoning_tokens']:5.0f}       (1.5B was 524)")
print(f"  truncation rate     {p['truncation_rate']*100:5.1f}%      (1.5B was 12.0%)")
print(f"  tokens/second       {p['tokens_per_second']:5.1f}       (1.5B was 91.5)")
print(f"  projected full run  {p['projected_phase3_hours']:5.1f} hours")
print(f"  gate passed         {p['passed']}")
print('=' * 58)
print('\nDECISION RULE')
print('  accuracy >= 65%  -> run the full 7B trace generation, 60-70% is reachable')
print('  accuracy 55-65%  -> borderline; worth one full run if GPU quota allows')
print('  accuracy <  55%  -> stay on 1.5B and spend the time on phases 7-8')
